cmake_minimum_required(VERSION 3.20)
project(neojson_packaging_consumer LANGUAGES CXX)

# Intentionally omit CMAKE_CXX_STANDARD to verify neojson::neojson propagates cxx_std_17.

option(NEOJSON_CONSUMER_USE_SUBDIRECTORY "Consume neojson via add_subdirectory instead of find_package" OFF)
set(NEOJSON_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../.." CACHE PATH "Path to neojson source directory")

if(NEOJSON_CONSUMER_USE_SUBDIRECTORY)
    add_subdirectory("${NEOJSON_SOURCE_DIR}" neojson_sub)
    if(TARGET neojson_tests)
        message(FATAL_ERROR "Expected NEOJSON_BUILD_TESTS to default OFF when consumed via add_subdirectory")
    endif()
    if(TARGET neojson_example_basic_parse_dump)
        message(FATAL_ERROR "Expected NEOJSON_BUILD_EXAMPLES to default OFF when consumed via add_subdirectory")
    endif()
    if(NEOJSON_INSTALL)
        message(FATAL_ERROR "Expected NEOJSON_INSTALL to default OFF when consumed via add_subdirectory")
    endif()
    get_target_property(_neojson_type neojson TYPE)
    if(WIN32)
        set(_expected_neojson_type "STATIC_LIBRARY")
    elseif(BUILD_SHARED_LIBS)
        set(_expected_neojson_type "SHARED_LIBRARY")
    else()
        set(_expected_neojson_type "STATIC_LIBRARY")
    endif()
    if(NOT _neojson_type STREQUAL "${_expected_neojson_type}")
        message(FATAL_ERROR "Expected neojson target type ${_expected_neojson_type}, got ${_neojson_type}")
    endif()
else()
    find_package(neojson 0.1 CONFIG REQUIRED)
endif()

add_executable(consumer consumer.cpp)
target_link_libraries(consumer PRIVATE neojson::neojson)

enable_testing()
add_test(NAME packaging_consumer COMMAND consumer)
